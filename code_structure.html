12. Go-Back-N - Server Logic

int expected = 0;

while (1) {

    recv(newsockfd,
         buffer,
         sizeof(buffer),
         0);

    frame = atoi(buffer);

    if (frame == expected) {

        printf("Frame %d received\n", frame);

        sprintf(buffer, "ACK %d", frame);

        send(newsockfd,
             buffer,
             strlen(buffer) + 1,
             0);

        expected++;
    }
    else {

        printf("Frame %d discarded\n", frame);

        sprintf(buffer,
                "ACK %d",
                expected - 1);

        send(newsockfd,
             buffer,
             strlen(buffer) + 1,
             0);
    }
}
